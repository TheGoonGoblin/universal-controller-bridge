cmake_minimum_required(VERSION 3.16)
project(UniversalControllerBridge VERSION 0.1.0 LANGUAGES CXX)
add_library(ucb INTERFACE)
target_include_directories(ucb INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(ucb INTERFACE cxx_std_17)
add_executable(ucb_decode src/main.cpp)
target_link_libraries(ucb_decode PRIVATE ucb)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_xbox360 tests/test_xbox360.cpp)
  target_link_libraries(test_xbox360 PRIVATE ucb)
  add_test(NAME xbox360 COMMAND test_xbox360)
endif()
