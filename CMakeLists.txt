cmake_minimum_required(VERSION 3.16)
project(UniversalControllerBridge VERSION 0.1.0 LANGUAGES CXX)
option(UCB_BUILD_USB "Build host libusb diagnostic (not a verified PS5 backend)" OFF)
add_library(ucb INTERFACE)
target_include_directories(ucb INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(ucb INTERFACE cxx_std_17)
add_executable(ucb_decode src/main.cpp)
target_link_libraries(ucb_decode PRIVATE ucb)
add_executable(ucb_app_preview tools/app_preview.cpp)
target_link_libraries(ucb_app_preview PRIVATE ucb)
if(UCB_BUILD_USB)
  find_path(UCB_LIBUSB_INCLUDE_DIR libusb.h PATH_SUFFIXES libusb-1.0)
  find_library(UCB_LIBUSB_LIBRARY NAMES usb-1.0 libusb-1.0)
  if(NOT UCB_LIBUSB_INCLUDE_DIR OR NOT UCB_LIBUSB_LIBRARY)
    message(FATAL_ERROR "Install libusb development files, or set UCB_LIBUSB_INCLUDE_DIR and UCB_LIBUSB_LIBRARY")
  endif()
  add_executable(ucb_usb tools/usb_diagnostic.cpp)
  target_include_directories(ucb_usb SYSTEM PRIVATE ${UCB_LIBUSB_INCLUDE_DIR})
  target_link_libraries(ucb_usb PRIVATE ucb ${UCB_LIBUSB_LIBRARY})
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_xbox360 tests/test_xbox360.cpp)
  target_link_libraries(test_xbox360 PRIVATE ucb)
  add_test(NAME xbox360 COMMAND test_xbox360)
  add_executable(test_integration tests/test_integration.cpp)
  target_link_libraries(test_integration PRIVATE ucb)
  add_test(NAME integration COMMAND test_integration)
  add_executable(test_app tests/test_app.cpp)
  target_link_libraries(test_app PRIVATE ucb)
  add_test(NAME app COMMAND test_app)
  if(UCB_BUILD_USB)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
      # Fake libusb symbols exercise failure and transfer paths without hardware.
      # This fixture is deliberately NOT linked to the real USB library.
      add_executable(ucb_usb_fixture tools/usb_diagnostic.cpp tests/fake_libusb.cpp)
      target_include_directories(ucb_usb_fixture SYSTEM PRIVATE ${UCB_LIBUSB_INCLUDE_DIR})
      target_link_libraries(ucb_usb_fixture PRIVATE ucb)
      add_test(NAME usb_fixture COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_usb_cli.py $<TARGET_FILE:ucb_usb_fixture>)
    endif()
  endif()
endif()
